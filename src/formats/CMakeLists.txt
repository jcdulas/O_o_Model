# Each adapter depends on the in-memory model. The core has no dependency on any adapter.
add_library(oor_api STATIC ReaderFiles.cpp)
add_library(oor::api ALIAS oor_api)
target_link_libraries(oor_api PUBLIC oor_core)
oor_enable_warnings(oor_api)

include(FetchContent)
foreach(format oos ooh5)
    if(format IN_LIST OOR_FORMATS)
        if(format STREQUAL "oos")
            set(adapter_revision a4d543c88037a3cbc7c30c5ace8ddc4959bba810)
        else()
            set(adapter_revision 1a39c47e8f57971ce36db49871c9887918447e9a)
        endif()
        string(TOUPPER "${format}" symbol)
        FetchContent_Declare(oor_${format}
            GIT_REPOSITORY https://github.com/jcdulas/O_o_${symbol}.git
            GIT_TAG ${adapter_revision})
        FetchContent_MakeAvailable(oor_${format})
        target_link_libraries(oor_api PUBLIC oor::${format})
        target_compile_definitions(oor_api PRIVATE OOR_WITH_${symbol})
    endif()
endforeach()

if("xlsx" IN_LIST OOR_FORMATS)
    add_library(oor_format_xlsx STATIC ../results/xlsx/XlsxWriter.cpp ../io/zip/Zip.cpp)
    add_library(oor::xlsx ALIAS oor_format_xlsx)
    target_link_libraries(oor_format_xlsx PUBLIC oor_core PRIVATE ZLIB::ZLIB)
    target_link_libraries(oor_api PUBLIC oor_format_xlsx)
    target_compile_definitions(oor_api PRIVATE OOR_WITH_XLSX)
    oor_enable_warnings(oor_format_xlsx)
endif()
